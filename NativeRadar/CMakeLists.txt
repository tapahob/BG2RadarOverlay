cmake_minimum_required(VERSION 3.16)
project(BG2Radar LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use a Windows x64 toolchain.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
                              GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
add_library(BG2Radar MODULE Radar.cpp CombatEvents.cpp)
target_compile_features(BG2Radar PRIVATE cxx_std_17)
target_compile_definitions(BG2Radar PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(BG2Radar PRIVATE minhook opengl32 gdi32 user32 d3d11 dcomp)
set_target_properties(BG2Radar PROPERTIES PREFIX "" SUFFIX ".asi")

option(BUILD_TESTING "Build native radar tests" OFF)
if(BUILD_TESTING)
  enable_testing()
  add_executable(CombatAmountTests tests/CombatAmountTests.cpp)
  target_compile_features(CombatAmountTests PRIVATE cxx_std_17)
  target_compile_definitions(CombatAmountTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(CombatAmountTests PRIVATE minhook)
  add_test(NAME CombatAmountTests COMMAND CombatAmountTests)
endif()

install(TARGETS BG2Radar LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION . RENAME MinHook-LICENSE.txt)
install(FILES README.md DESTINATION .)
install(FILES ../EEexMod/M_RDRCRT.lua DESTINATION .)
